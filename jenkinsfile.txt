pipeline {

    agent any

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build') {
            steps {
                bat 'mvn clean compile'
            }
        }

        stage('Run Automation') {
            steps {
                bat 'mvn test'
            }
        }
    }

    post {

        always {
            echo 'Automation execution completed'
        }

        success {
            echo 'Automation execution passed'
        }

        failure {
            echo 'Automation execution failed'
        }
    }
}
